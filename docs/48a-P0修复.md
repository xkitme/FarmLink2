#### 1.3 `backend/src/middleware/auth.js` —— 校验 token iat 不早于 passwordChangedAt

读 token 后查 user，比较 `decoded.iat * 1000 < user.passwordChangedAt.getTime()` → 抛 401「请重新登录」。

具体改：

```js
import jwt from 'jsonwebtoken'
import { prisma } from '../db.js'
import { config } from '../config/index.js'
import { errors } from '../utils/response.js'

export async function requireAuth(req, res, next) {
  try {
    const header = req.headers.authorization || ''
    const token = header.startsWith('Bearer ') ? header.slice(7) : null
    if (!token) throw errors.unauthorized()
    const decoded = jwt.verify(token, config.jwtSecret)
    const user = await prisma.user.findUnique({
      where: { id: decoded.id },
      select: { id: true, role: true, regionCode: true, status: true, passwordChangedAt: true },
    })
    if (!user || user.status !== 1) throw errors.unauthorized('账号不可用')

    // G1：token iat 必须晚于 passwordChangedAt
    if (user.passwordChangedAt &&
        decoded.iat * 1000 < user.passwordChangedAt.getTime() - 1000 /* 1s 容差 */) {
      throw errors.unauthorized('密码已修改，请重新登录')
    }
    req.user = user
    next()
  } catch (e) {
    if (e.name === 'JsonWebTokenError' || e.name === 'TokenExpiredError') {
      return next(errors.unauthorized())
    }
    next(e)
  }
}
```

#### 3.4 后端 `qaRecords` —— 列表 thread 化分组

`qaRecords` 改为返回每个 thread 的「首条 record + 该 thread 的消息数」：

```js
export async function qaRecords(req, res) {
  const where = req.user.role === 'ADMIN' && req.query.userId
    ? { userId: Number(req.query.userId) } : { userId: req.user.id }
  // 取每个 thread 的最早一条
  const all = await prisma.aiQaRecord.findMany({
    where, orderBy: { createdAt: 'asc' },
    select: { id: true, threadId: true, scene: true, question: true, answer: true, createdAt: true, referencesJson: true },
  })
  // 按 threadId 分组，取首条 + count
  const grouped = new Map()
  for (const r of all) {
    const t = r.threadId || r.id
    if (!grouped.has(t)) grouped.set(t, { ...r, threadId: t, messageCount: 1 })
    else grouped.get(t).messageCount += 1
  }
  // 按首条时间倒序
  const threads = [...grouped.values()].sort(
    (a, b) => new Date(b.createdAt) - new Date(a.createdAt)
  )
  // 分页处理略
  okPage(res, { records: threads, total: threads.length, pageNum: 1, pageSize: threads.length })
}
```
